set(CMAKE_CONFIGURATION_TYPES Debug Release RelWithDebInfo MinSizeRel)

if(BUILD_KIT)
	message(NOTICE "Build Kit is defined as ${BUILD_KIT}")
endif()

message(NOTICE "CMAKE_CURRENT_BINARY_DIR is defined as ${CMAKE_CURRENT_BINARY_DIR}")

	if(DEFINED CMAKE_BUILD_TYPE)

cmake_minimum_required(VERSION 3.16.4)

# e2 studio's CMake build configuration passes no toolchain file, so CMake would
# pick arm-none-eabi-gcc from PATH, which rejects the LLVM flags generated by RASC.
if(NOT CMAKE_TOOLCHAIN_FILE)
	set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/cmake/llvm.cmake" CACHE FILEPATH "Toolchain file")
endif()

# CMake 3.27 (bundled with e2 studio) writes a backslash toolchain path unescaped
# into CMakeCCompiler.cmake and then fails with "Invalid character escape '\U'".
foreach(_tc_env ARM_LLVM_TOOLCHAIN_PATH ARM_TOOLCHAIN_PATH)
	if(DEFINED ENV{${_tc_env}})
		file(TO_CMAKE_PATH "$ENV{${_tc_env}}" _tc_path)
		set(ENV{${_tc_env}} "${_tc_path}")
	endif()
endforeach()

# Project configuration
# The display name ("OTA Update over BLE", also the Eclipse/.project name) cannot be
# the CMake project name: PROJECT_NAME becomes the target "${PROJECT_NAME}.elf" in
# the RASC-generated GeneratedSrc.cmake, and CMake rejects target names with spaces.
# PROJECT_NAME therefore drives the image names: ota_update_over_ble.elf/.hex/.map.
project(ota_update_over_ble
	VERSION 1.0.0
	DESCRIPTION "OTA Update over BLE"
	LANGUAGES C CXX ASM)

set(CMAKE_CROSSCOMPILING "TRUE")
set(CMAKE_EXPORT_COMPILE_COMMANDS "TRUE")

include(${CMAKE_CURRENT_LIST_DIR}/Config.cmake)

include(${CMAKE_CURRENT_LIST_DIR}/cmake/GeneratedCfg.cmake)

# If required user can amend RASC generated flags here, e.g. list(REMOVE_ITEM RASC_CMAKE_C_FLAGS "-Wall")

# ---------------------------------------------------------------------------------
# Optimisation: -Os in every configuration (the former managed build used
# "codeSize"; RASC defaults to -O2). DWARF 4 for the Renesas debugger.
# ---------------------------------------------------------------------------------
foreach(_rasc_opt_var RASC_DEBUG_FLAGS RASC_RELEASE_FLAGS RASC_MIN_SIZE_RELEASE_FLAGS RASC_RELEASE_WITH_DEBUG_INFO)
	list(TRANSFORM ${_rasc_opt_var} REPLACE "^-O[0-3sgz]?$" "-Os")
endforeach()
list(APPEND RASC_DEBUG_FLAGS "-gdwarf-4")
list(APPEND RASC_RELEASE_WITH_DEBUG_INFO "-gdwarf-4")

# ---------------------------------------------------------------------------------
# OTA linker script: the app is linked at the primary slot (0x00010000) with the
# bootloader embedded at 0x0, instead of the stock script/fsp.lld.
# ---------------------------------------------------------------------------------
set(OTA_LINKER_SCRIPT "${CMAKE_CURRENT_LIST_DIR}/script/ota.lld")
list(FIND RASC_CMAKE_EXE_LINKER_FLAGS "-T" _t_idx)
if(_t_idx GREATER -1)
	math(EXPR _t_arg "${_t_idx} + 1")
	list(REMOVE_AT RASC_CMAKE_EXE_LINKER_FLAGS ${_t_idx} ${_t_arg})
endif()
list(FILTER RASC_CMAKE_EXE_LINKER_FLAGS EXCLUDE REGEX "^-T.+")
list(APPEND RASC_CMAKE_EXE_LINKER_FLAGS "-T" "${OTA_LINKER_SCRIPT}")

include(${CMAKE_CURRENT_LIST_DIR}/cmake/GeneratedSrc.cmake)

set(APP_TARGET ${PROJECT_NAME}.elf)

# Header folders under src/ (sources are already globbed by GeneratedSrc.cmake).
target_include_directories(${APP_TARGET}
	PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/include
	${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/src
	${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/src/hal
	${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/ble
	${CMAKE_CURRENT_SOURCE_DIR}/src/segger_rtt/RTT
	${CMAKE_CURRENT_SOURCE_DIR}/src/segger_rtt/Config
)

# Vendor code (DA14xxx GTL driver, TinyCrypt, FSP GTL BLE abstraction) does not
# survive RASC's strict warning flags (-Wconversion/-Wextra). Silence it so the
# build output only shows warnings from first-party src/.
file(GLOB_RECURSE Vendor_Source_Files
	${CMAKE_CURRENT_SOURCE_DIR}/ra/renesas/*.c
	${CMAKE_CURRENT_SOURCE_DIR}/ra/intel/*.c
	${CMAKE_CURRENT_SOURCE_DIR}/ra/fsp/src/rm_ble_abs_gtl/*.c
	${CMAKE_CURRENT_SOURCE_DIR}/src/segger_rtt/RTT/*.c)
if(Vendor_Source_Files)
	set_source_files_properties(${Vendor_Source_Files} PROPERTIES COMPILE_FLAGS "-w")
endif()

# Embedded bootloader: ".incbin ../src/rs_ota_fw_update_over_ble/bootloader.bin"
# resolves through the -I <project>/src include path. .incbin leaves no
# dependency information, so state it explicitly.
set_property(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/src/hal/rs_ota_fw_update_over_ble_boot_rsboot.c"
	APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/rs_ota_fw_update_over_ble/bootloader.bin")
set_property(TARGET ${APP_TARGET} APPEND PROPERTY LINK_DEPENDS "${OTA_LINKER_SCRIPT}")

# ---------------------------------------------------------------------------------
# OTA artifacts (port of the former makefile.targets)
#   <name>_ota.bin - raw primary-slot image to send over SUOTA; bootloader, option
#                    settings and data flash stripped; must fit the 96 KB slot
#   <name>_ota.map - linker map of exactly this build
# ---------------------------------------------------------------------------------
get_filename_component(_tc_bin_dir "${CMAKE_C_COMPILER}" DIRECTORY)
find_program(OTA_OBJCOPY NAMES llvm-objcopy HINTS "${_tc_bin_dir}" NO_DEFAULT_PATH)
find_program(OTA_SIZE    NAMES llvm-size    HINTS "${_tc_bin_dir}" NO_DEFAULT_PATH)
set(OTA_IMAGE_NAME ${PROJECT_NAME}_ota)
set(OTA_SLOT_SIZE  0x18000)   # BOOT_PRIMARY_SIZE / FLASH_LENGTH in script/ota.lld

add_custom_command(TARGET ${APP_TARGET} POST_BUILD
	COMMAND ${OTA_OBJCOPY} -O binary -w --gap-fill 0xFF
	        -R .bootloader -R .option_setting* -R __option_setting* -R *data_flash*
	        ${APP_TARGET} ${OTA_IMAGE_NAME}.bin
	COMMAND ${CMAKE_COMMAND} -E copy_if_different ${PROJECT_NAME}.map ${OTA_IMAGE_NAME}.map
	COMMAND ${CMAKE_COMMAND} -DBIN=${OTA_IMAGE_NAME}.bin -DSLOT_SIZE=${OTA_SLOT_SIZE}
	        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/ota_size_check.cmake
	COMMAND ${OTA_SIZE} --format=berkeley ${APP_TARGET}
	BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/${OTA_IMAGE_NAME}.bin ${CMAKE_CURRENT_BINARY_DIR}/${OTA_IMAGE_NAME}.map
	WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
	COMMENT "Creating ${OTA_IMAGE_NAME}.bin / ${OTA_IMAGE_NAME}.map"
	VERBATIM)

# Print image size and OTA slot usage on EVERY build, also when nothing had to be
# relinked ("ninja: no work to do"), so the IDE console always shows them.
add_custom_target(size_report ALL
	COMMAND ${CMAKE_COMMAND} -DBIN=${OTA_IMAGE_NAME}.bin -DSLOT_SIZE=${OTA_SLOT_SIZE}
	        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/ota_size_check.cmake
	COMMAND ${OTA_SIZE} --format=berkeley ${APP_TARGET}
	DEPENDS ${APP_TARGET}
	WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
	VERBATIM)
